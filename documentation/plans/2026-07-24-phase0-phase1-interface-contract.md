## 2. RUST CORE PUBLIC API (`photoscore/src/lib.rs`)

All types below are `#[derive(uniffi::Record)]`, `#[derive(uniffi::Enum)]`, `#[derive(uniffi::Error)]`, or `#[derive(uniffi::Object)]` as annotated. Field names and function names are FINAL. Swift names are the auto-lowerCamelCased equivalents (noted where non-obvious).

### 2.1 Value types (`models` crate, re-exported from `photoscore`)

```rust
/// Which Synology namespace family a query targets.
#[derive(uniffi::Enum, Clone, Copy, Debug, PartialEq, Eq)]
pub enum Space {
    Personal, // -> SYNO.Foto.*
    Shared,   // -> SYNO.FotoTeam.*
}

#[derive(uniffi::Enum, Clone, Copy, Debug, PartialEq, Eq)]
pub enum MediaKind {
    Photo,
    Video,
    Unknown, // tolerant-decode fallback; never fail on an unrecognized type
}

/// Immutable connection parameters. No secrets stored here beyond the login call.
#[derive(uniffi::Record, Clone, Debug)]
pub struct Connection {
    pub host: String,       // "https://192.168.1.10:5001" or tailscale name; scheme required
    pub verify_tls: bool,   // true normally; probe/dev path may pin a cert instead (see 2.6)
    pub pinned_cert_der: Option<Vec<u8>>, // DER cert to trust for THIS host only; None = system roots
}

/// Result of a successful login. `sid` is what Swift stores in Keychain.
#[derive(uniffi::Record, Clone, Debug)]
pub struct Session {
    pub sid: String,
    pub syno_token: Option<String>, // present when DSM returns SynoToken; sent on later calls
    pub username: String,
    pub device_did: Option<String>, // "remember device" token if issued; else None
}

/// One media item. Metadata only; bytes fetched separately.
#[derive(uniffi::Record, Clone, Debug)]
pub struct Asset {
    pub id: i64,             // server item id (identity within a space)
    pub cache_key: String,   // version token from server; NOT identity
    pub filename: String,
    pub media_kind: MediaKind,
    pub taken_at: Option<i64>,   // unix epoch seconds, UTC; None if server omits
    pub added_at: Option<i64>,   // unix epoch seconds, UTC
    pub width: Option<u32>,
    pub height: Option<u32>,
    pub file_size: Option<u64>,  // bytes
    pub space: Space,
    pub server_version: Option<i64>, // for delta reconciliation; None if server omits
}

#[derive(uniffi::Record, Clone, Debug)]
pub struct Album {
    pub id: i64,
    pub name: String,
    pub item_count: u32,
    pub cover_cache_key: Option<String>,
    pub space: Space,
}

/// Emitted during crawl. `complete` flips true only at the barrier.
#[derive(uniffi::Record, Clone, Debug)]
pub struct CrawlProgress {
    pub space: Space,
    pub done: u64,      // items persisted so far
    pub total: u64,     // expected total reported by server (0 if not yet known)
    pub complete: bool, // initial_crawl_complete barrier reached for this space
}

/// One capability discovered via SYNO.API.Info.
#[derive(uniffi::Record, Clone, Debug)]
pub struct ApiCapability {
    pub name: String,        // e.g. "SYNO.Foto.Browse.Item"
    pub path: String,        // cgi path, e.g. "entry.cgi"
    pub min_version: u32,
    pub max_version: u32,
}

/// A downloaded thumbnail: bytes plus the on-disk path the core cached them at.
#[derive(uniffi::Record, Clone, Debug)]
pub struct ThumbnailData {
    pub cached_path: String, // absolute path on disk; stable for (asset_id,size,cache_key)
    pub bytes: Vec<u8>,      // the image bytes (Swift may ignore and read path)
}

#[derive(uniffi::Enum, Clone, Copy, Debug, PartialEq, Eq)]
pub enum ThumbnailSize {
    Sm, // 240
    M,  // 320
    Xl, // 1280
}

/// Auth phase surfaced to Swift's state machine.
#[derive(uniffi::Enum, Clone, Copy, Debug, PartialEq, Eq)]
pub enum SessionState {
    Valid,
    Expired,   // SID rejected; re-login without necessarily re-prompting 2FA if device_did held
    Invalid,   // credentials bad
}
```

### 2.2 Error type (fail-closed contract)

```rust
#[derive(uniffi::Error, Debug, thiserror::Error)]
pub enum CoreError {
    /// Credentials or SID rejected. Swift -> Invalid/Expired.
    #[error("authentication failed: {message}")]
    Auth { message: String },

    /// 2FA required but no otp_code supplied, or otp_code wrong.
    #[error("two-factor code required or incorrect")]
    OtpRequired,

    /// Reachability / TLS / timeout. Retryable.
    #[error("network error: {message}")]
    Network { message: String },

    /// Body did not decode even under tolerant rules (not merely unknown fields).
    #[error("decode error: {message}")]
    Decode { message: String },

    /// Server returned a shape/version/code we do not handle -> FAIL CLOSED.
    #[error("unexpected server response: {message}")]
    UnexpectedResponse { message: String },

    /// A write/mutation was attempted in a read-only build. FAIL CLOSED. Never reached in P0/P1.
    #[error("write refused: read-only mode")]
    WriteRefused,

    /// Local persistence failure.
    #[error("storage error: {message}")]
    Storage { message: String },

    /// Requested API/version not offered by this NAS (capability probe gate).
    #[error("capability unavailable: {api}")]
    CapabilityUnavailable { api: String },
}
```

Mapping rule (authoritative): Synology `error.code` 400/401/403/booked-2FA codes (400 = no such account/pw, 403 = 2FA needed, 404 = OTP wrong on many DSM builds) map to `Auth`/`OtpRequired`. Transport failures map to `Network`. Serde failure after tolerant decode maps to `Decode`. `success:false` with an unknown code maps to `UnexpectedResponse`. Never `panic!` across the FFI boundary; every public method returns `Result<_, CoreError>`.

### 2.3 The exposed object

```rust
/// The single UniFFI-exported facade. Swift holds one instance per app run.
#[derive(uniffi::Object)]
pub struct PhotosCore { /* holds Store, ApiClient (post-login), config; interior mutability */ }

#[uniffi::export(async_runtime = "tokio")]
impl PhotosCore {
    /// Construct with a local DB directory. Opens/creates SQLite + runs migrations.
    /// Not async: pure local setup. Fallible -> Storage.
    #[uniffi::constructor]
    pub fn new(db_dir: String, cache_dir: String) -> Result<Arc<Self>, CoreError>;

    // ---- AUTH ----

    /// Log in. otp_code None on first attempt; if OtpRequired returned, Swift re-calls with Some.
    pub async fn login(
        &self,
        connection: Connection,
        username: String,
        password: String,
        otp_code: Option<String>,
    ) -> Result<Session, CoreError>;

    /// Restore a prior session (SID from Keychain) without re-entering credentials.
    /// Validates by a cheap authed call; returns state.
    pub async fn restore_session(
        &self,
        connection: Connection,
        session: Session,
    ) -> Result<SessionState, CoreError>;

    /// Clean teardown: server logout, drop in-memory client, clear per-account local cache.
    /// Idempotent. Swift additionally clears Keychain SID. Never throws on already-signed-out.
    pub async fn sign_out(&self) -> Result<(), CoreError>;

    // ---- CAPABILITY ----

    /// SYNO.API.Info query=all. Pins versions used by later calls. Must be called after login,
    /// before crawl. Returns the discovered set (also cached internally).
    pub async fn probe_capabilities(&self) -> Result<Vec<ApiCapability>, CoreError>;

    // ---- CRAWL / SYNC ----

    /// Resumable initial crawl for one space. Persists page cursor; safe to re-call to resume.
    /// Streams progress via the observer. Returns final CrawlProgress with complete=true.
    /// FfiCrawlObserver is a UniFFI callback interface (see 2.4).
    pub async fn crawl_space(
        &self,
        space: Space,
        observer: Box<dyn FfiCrawlObserver>,
    ) -> Result<CrawlProgress, CoreError>;

    /// Delta reconciliation by server (id, version). Call after initial crawl is complete.
    pub async fn reconcile_delta(&self, space: Space) -> Result<CrawlProgress, CoreError>;

    /// Read the persisted barrier/progress without doing network work.
    pub fn crawl_progress(&self, space: Space) -> Result<CrawlProgress, CoreError>;

    // ---- WINDOWED READ (local DB only, no network) ----

    /// Windowed slice for the grid, newest-first by taken_at then id. Never returns partial-crawl
    /// data as if complete: caller gates on crawl_progress().complete.
    pub fn fetch_assets(
        &self,
        space: Space,
        offset: u32,
        limit: u32,
    ) -> Result<Vec<Asset>, CoreError>;

    /// Total local asset count for a space.
    pub fn asset_count(&self, space: Space) -> Result<u64, CoreError>;

    pub fn fetch_albums(&self, space: Space) -> Result<Vec<Album>, CoreError>;

    // ---- MEDIA BYTES (network + disk cache) ----

    /// Fetch a thumbnail. Returns cached path + bytes. Cache key is composite
    /// (asset_id, size, cache_key); a changed cache_key invalidates the old entry.
    pub async fn thumbnail(
        &self,
        space: Space,
        asset_id: i64,
        cache_key: String,
        size: ThumbnailSize,
    ) -> Result<ThumbnailData, CoreError>;

    /// Download the ORIGINAL to a temp file. Returns absolute path. Read-only; never mutates NAS.
    pub async fn download_original(
        &self,
        space: Space,
        asset_id: i64,
        cache_key: String,
    ) -> Result<String, CoreError>;
}
```

### 2.5 Swift-side names (auto-generated, for author reference)

`PhotosCore(dbDir:cacheDir:)`, `try await core.login(connection:username:password:otpCode:)`, `restoreSession`, `signOut`, `probeCapabilities`, `crawlSpace(space:observer:)`, `reconcileDelta(space:)`, `crawlProgress(space:)`, `fetchAssets(space:offset:limit:)`, `assetCount(space:)`, `fetchAlbums(space:)`, `thumbnail(space:assetId:cacheKey:size:)`, `downloadOriginal(space:assetId:cacheKey:)`. ### 2.6 TLS trust contract (locked)

Global TLS validation is NEVER disabled. `Connection.verify_tls` defaults true using system roots. For the DSM self-signed cert reachable over a Tailscale name that won't match the cert CN, the ONLY sanctioned path is `pinned_cert_der: Some(der)`: reqwest is built with `.add_root_certificate()` + `.danger_accept_invalid_hostnames(false)` scoped to that one client, trusting exactly that cert. `danger_accept_invalid_certs` is forbidden in any committed code path. Phase 0 has a task to capture the DER and document the pinning decision.

**Linking into Xcode:** the app target adds `PhotosCore.xcframework` under "Frameworks, Libraries, and Embedded Content"; `bindings/PhotosCore.swift` is added to the app target's compile sources; `bindings/module.modulemap` + `PhotosCoreFFI.h` exposed via a header search path / the framework's module. A pre-build Run Script phase runs `make bindings` (or authors run it manually and commit `bindings/`). The committed `bindings/PhotosCore.swift` is the source of truth Swift authors import via `import PhotosCore`.

**Async across the boundary:** `#[uniffi::export(async_runtime = "tokio")]` makes each `pub async fn` a Swift `async throws` method. The core owns a Tokio multi-thread runtime; Swift calls with `try await`. No blocking calls on the main thread from Swift.

**Errors across the boundary:** `CoreError` (`#[derive(uniffi::Error)]`) becomes a Swift `enum CoreError: Error` with associated values matching the Rust fields (e.g. `.auth(message: String)`, `.otpRequired`, `.unexpectedResponse(message:)`). Swift authors `catch` on it in `PhotosCoreClient`.

**Callback:** `FfiCrawlObserver` (callback interface) lets Rust call back into a Swift class conforming to the generated `FfiCrawlObserver` protocol during `crawlSpace`.

## 4. SQLITE SCHEMA (`core/persistence/src/schema.rs`, embedded DDL)

`space` stored as INTEGER (0=Personal, 1=Shared). Identity of an asset is `(space, server_id)`. Thumbnail cache identity is the composite `(server_id, size_variant, cache_key)` (asset_id + size + version token), enforced by the on-disk cache naming; the DB stores `cache_key` per asset so a change invalidates.

```sql
PRAGMA journal_mode = WAL;
PRAGMA foreign_keys = ON;

CREATE TABLE IF NOT EXISTS assets (
    rowid_pk       INTEGER PRIMARY KEY AUTOINCREMENT,
    space          INTEGER NOT NULL,            -- 0 Personal, 1 Shared
    server_id      INTEGER NOT NULL,            -- Synology item id (identity within space)
    cache_key      TEXT    NOT NULL,            -- version token, NOT identity
    filename       TEXT    NOT NULL,
    media_kind     INTEGER NOT NULL DEFAULT 2,  -- 0 photo, 1 video, 2 unknown
    taken_at       INTEGER,                     -- unix epoch seconds UTC, nullable
    added_at       INTEGER,                     -- unix epoch seconds UTC, nullable
    width          INTEGER,
    height         INTEGER,
    file_size      INTEGER,
    server_version INTEGER,                     -- for delta reconciliation, nullable
    updated_at     INTEGER NOT NULL,            -- local write time, epoch seconds
    UNIQUE (space, server_id)
);
CREATE INDEX IF NOT EXISTS idx_assets_space_taken
    ON assets (space, taken_at DESC, server_id DESC);   -- grid ordering
CREATE INDEX IF NOT EXISTS idx_assets_space_ver
    ON assets (space, server_id, server_version);       -- delta reconcile

CREATE TABLE IF NOT EXISTS albums (
    rowid_pk        INTEGER PRIMARY KEY AUTOINCREMENT,
    space           INTEGER NOT NULL,
    server_id       INTEGER NOT NULL,
    name            TEXT    NOT NULL,
    item_count      INTEGER NOT NULL DEFAULT 0,
    cover_cache_key TEXT,
    updated_at      INTEGER NOT NULL,
    UNIQUE (space, server_id)
);
CREATE INDEX IF NOT EXISTS idx_albums_space ON albums (space, name);

-- exactly one row per space; tracks resumable crawl + barrier
CREATE TABLE IF NOT EXISTS sync_state (
    space                  INTEGER PRIMARY KEY,  -- 0 Personal, 1 Shared (one row each)
    initial_crawl_complete INTEGER NOT NULL DEFAULT 0,  -- 0/1 barrier
    expected_total         INTEGER NOT NULL DEFAULT 0,  -- server-reported total for the space
    last_offset            INTEGER NOT NULL DEFAULT 0,  -- resumable page cursor (offset)
    last_page_limit        INTEGER NOT NULL DEFAULT 0,  -- page size used, for resume sanity
    highest_seen_version   INTEGER,                     -- delta high-water mark, nullable
    last_crawl_at          INTEGER,                     -- epoch seconds of last completed pass
    last_reconcile_at      INTEGER                      -- epoch seconds of last delta pass
);

-- schema versioning for the migration runner
CREATE TABLE IF NOT EXISTS schema_meta (
    key   TEXT PRIMARY KEY,
    value TEXT NOT NULL
);
-- seeded: ('schema_version','1')
```

Rules: `fetch_assets` orders by `idx_assets_space_taken` (NULL `taken_at` sorts last via `taken_at DESC` with a NULLS-last shim in the query). Crawl upserts on `(space, server_id)`. Barrier flip: `initial_crawl_complete` set to 1 only when `last_offset >= expected_total` and the final page returned fewer than `last_page_limit`. On-disk thumbnail file path (owned by core, not the DB): `{cache_dir}/thumbs/{space}/{server_id}/{size}_{cache_key}.jpg`; a stale `cache_key` yields a new filename, orphaning the old (swept later).
